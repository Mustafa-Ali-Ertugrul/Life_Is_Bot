function addHabit(user, name, days, times):

    habit = createHabit()
    habit.user_id = user.id
    habit.name = name
    habit.is_active = true
    save(habit)

    for each day in days:
        for each time in times:
            schedule = createHabitSchedule()
            schedule.habit_id = habit.id
            schedule.days_of_week = day
            schedule.time_of_day = time
            schedule.is_active = true
            save(schedule)

    sendConfirmation(user, "Rutin eklendi.")
