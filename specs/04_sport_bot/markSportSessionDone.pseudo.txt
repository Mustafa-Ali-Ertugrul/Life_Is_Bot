function markSportSessionDone(user, sessionId, status):

    session = getSportSession(sessionId)

    if session.user_id != user.id:
        reject()

    log = createSportLog()

    log.sport_session_id = session.id
    log.user_id = user.id
    log.completed_at = now()
    log.status = status

    save(log)

    session.status = status
    save(session)
